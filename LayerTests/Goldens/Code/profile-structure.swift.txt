import SwiftUI

struct Profile: View {
    var body: some View {
        VStack(alignment: .leading, spacing: DesignTokens.rhythm[3]) {
            Text("Profile")
                .font(.title.bold())
                .multilineTextAlignment(.center)
            HStack(alignment: .center, spacing: DesignTokens.rhythm[1]) {
                Image("avatar")
                    .resizable()
                    .scaledToFill()
                    .frame(width: 48, height: 48)
                    .clipShape(.circle)
                Text("Alex Tan")
                    .font(.body)
            }
                .frame(maxWidth: .infinity, alignment: .leading)
            VStack(alignment: .leading, spacing: DesignTokens.rhythm[0]) {
                VStack(alignment: .leading, spacing: DesignTokens.rhythm[0]) {
                    Text("About me")
                        .font(.body)
                    Text("Designing better experiences.")
                        .font(.subheadline)
                        .multilineTextAlignment(.center)
                        .frame(maxWidth: .infinity)
                }
                    .frame(maxWidth: .infinity, alignment: .leading)
            }
                .padding(DesignTokens.cardPadding)
                .frame(maxWidth: .infinity, alignment: .leading)
                .background(.background.secondary, in: .rect(cornerRadius: DesignTokens.cardRadius))
            Grid(horizontalSpacing: DesignTokens.rhythm[1], verticalSpacing: DesignTokens.rhythm[1]) {
                GridRow {
                    Text("Posts")
                        .font(.body)
                        .frame(maxWidth: .infinity)
                    Text("Followers")
                        .font(.body)
                        .frame(maxWidth: .infinity)
                }
                GridRow {
                    Text("Following")
                        .font(.body)
                        .frame(maxWidth: .infinity)
                    Text("Likes")
                        .font(.body)
                        .frame(maxWidth: .infinity)
                }
            }
                .frame(maxWidth: .infinity)
            Button {
            } label: {
                Text("Edit profile")
                    .font(.headline)
                    .foregroundStyle(.white)
                    .frame(height: 50)
                    .frame(maxWidth: .infinity)
                    .background(Color(red: 0.80, green: 0.42, blue: 0.34), in: .capsule)
            }
        }
        .padding(DesignTokens.margin)
        .frame(maxWidth: .infinity, alignment: .leading)
        .frame(maxHeight: .infinity, alignment: .top)
    }
}

/// The numbers this design is laid out and styled by.
enum DesignTokens {
    /// The steps the space between things is made of, smallest first.
    static let rhythm: [CGFloat] = [8, 12, 16, 24, 32]
    static let cornerRadiusSmall: CGFloat = 8
    static let cornerRadiusMedium: CGFloat = 12
    static let cornerRadiusLarge: CGFloat = 16
    static let cardPadding: CGFloat = 16
    static let cardRadius: CGFloat = 16
    static let cardFill = Color(red: 0.95, green: 0.95, blue: 0.97)
    static let titleSize: CGFloat = 28
    static let bodySize: CGFloat = 15
    static let textSize: CGFloat = 17
    /// The space kept round a screen's column, inside its safe area.
    static let margin: CGFloat = 20
}
